\portada{8}{Taller integrador: de los datos a una decisión}

\begin{frame}{Objetivo del módulo}

\ruta{
\text{problema}\fl\text{datos}\fl\text{preparación}\fl\text{análisis}\fl\text{modelamiento}\fl\text{pronóstico}\fl\text{anomalías}\fl\text{decisión}
}
\idea{¿Qué dicen los datos, qué tan confiable es el resultado y qué recomendaría hacer?}
El objetivo no es ejecutar muchas funciones, sino \textbf{decidir e interpretar}.
\end{frame}

\begin{frame}{Del ejercicio académico al problema profesional}
En la industria nadie pide "calcule la ACF". Piden:
\begin{itemize}
  \item "creemos que el proceso se está comportando distinto";
  \item "necesitamos anticipar esta variable";
  \item "tenemos meses de datos, ¿se puede hacer algo útil?"
\end{itemize}
\vfill
Mala pregunta: \emph{¿cuál es el mejor ARIMA?}\\
Buena pregunta: \emph{¿con qué precisión podemos anticipar el tonelaje las próximas 8 h?}
\end{frame}

\begin{frame}{Paso 0 — Comprender antes de programar}
Antes de ejecutar Python, escribir:
\begin{itemize}
  \item Variable analizada, unidad, frecuencia temporal, periodo cubierto;
  \item Pregunta operacional concreta.
\end{itemize}
\vfill
Y para la limpieza: documentar \textbf{qué se hizo y por qué}.
\idea{dato raro $\neq$ dato incorrecto}
\end{frame}

\begin{frame}{De una métrica a una recomendación}
\[
MAE=50\ \text{t/h} \quad\text{con variable} \sim 1500\ \text{t/h}
\quad\Rightarrow\quad \approx 3.3\%
\]
\begin{itemize}
  \item \textbf{Dato}: serie horaria de producción;
  \item \textbf{Métrica}: $MAE=45$ t/h; supera al naïve en $\sim25\%$;
  \item \textbf{Hallazgo}: hay estructura temporal aprovechable;
  \item \textbf{Interpretación}: pronóstico de corto plazo viable;
  \item \textbf{Acción}: apoyo a planificación y monitoreo, con supervisión fuera de banda.
\end{itemize}
\end{frame}

\begin{frame}{La lógica de justificación}
\[
\boxed{\text{evidencia}\fl\text{decisión}}
\]
\begin{itemize}
  \item ACF con picos en múltiplos de 24 $\rightarrow$ se consideró periodicidad diaria;
  \item ADF no indicó estacionariedad $\rightarrow$ se aplicó diferenciación;
  \item el modelo A presentó autocorrelación residual $\rightarrow$ se descartó;
  \item el modelo B superó al naïve $\rightarrow$ se seleccionó;
  \item residuos persistentes $\rightarrow$ posible cambio de régimen.
\end{itemize}
\end{frame}

\begin{frame}{Errores a evitar}
\begin{columns}[T]
\column{0.5\linewidth}
\begin{itemize}
  \item modelar sin mirar los datos;
  \item eliminar outliers automáticamente;
  \item train/test aleatorio;
  \item elegir modelo solo por AIC.
\end{itemize}
\column{0.5\linewidth}
\begin{itemize}
  \item reportar únicamente métricas;
  \item confundir anomalía con causa;
  \item forecast sin incertidumbre;
  \item recomendar lo que los datos no respaldan.
\end{itemize}
\end{columns}
\end{frame}

\end{frame}
